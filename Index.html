import express from 'express'; import dotenv from 'dotenv'; import path from 'path'; import { fileURLToPath } from 'url'; import TelegramBot from 'node-telegram-bot-api'; import { requestLoanDisbursement } from './momoServer.js'; dotenv.config(); const app = report = express(); const PORT = process.env.PORT || 3000; const __filename = fileURLToPath(import.meta.url); const __dirname = path.dirname(__filename); app.use(express.json()); app.use(express.static(path.join(__dirname, 'public'))); // In-Memory cache dictionary pairing active session references const pendingLoansCache = new Map(); // Instantiate operational connection pools to Telegram APIs const bot = new TelegramBot(process.env.TELEGRAM_BOT_TOKEN, { polling: true }); const ADMIN_CHAT_ID = process.env.TELEGRAM_ADMIN_CHAT_ID; /** * Route: Receives User Application data from front-end layout template */ app.post('/api/loans/apply', async (req, res) => { const { fullName, mtnNumber, loanAmount, repaymentPeriodMonths, monthlyIncome, intendedPurpose } = req.body; // Structural constraint gatekeepers if (!fullName || !mtnNumber || !loanAmount || !repaymentPeriodMonths || !monthlyIncome || !intendedPurpose) { return res.status(400).json({ success: false, error: "Validation Failure: All requested parameters are mandatory." }); } const months = parseInt(repaymentPeriodMonths); if (isNaN(months) || months < 1 || months > 24) { return res.status(400).json({ success: false, error: "Invalid Parameter: Repayment horizon ceiling restriction violated." }); } // Build internal reference tracker key const applicationId = 'LN-' + Math.floor(100000 + Math.random() * 900000); // Explicitly cache details for transactional mapping records pendingLoansCache.set(applicationId, { fullName, mtnNumber, loanAmount, months, monthlyIncome, intendedPurpose, status: 'PENDING' }); // Comprehensive Structured Layout sent downstream to Telegram Admin chat const fullMessagePayload = ` 📋 *NEW LOAN APPLICATION DETECTED* ▬▬▬▬▬▬▬▬▬▬▬▬▬▬▬▬▬▬▬▬▬▬ 👤 *Applicant Name:* \`${fullName}\` 📞 *MTN Wallet Line:* \`${mtnNumber}\` 💰 *Principal Amount:* UGX ${parseFloat(loanAmount).toLocaleString()} 📅 *Horizon Duration:* ${months} Months 📈 *Monthly Income:* UGX ${parseFloat(monthlyIncome).toLocaleString()} 🌾 *Intended Vector:* ${intendedPurpose.toUpperCase()} 🆔 *Application Ref:* \`${applicationId}\` ▬▬▬▬▬▬▬▬▬▬▬▬▬▬▬▬▬▬▬▬▬▬ *Awaiting review verification actions...* `; const interactiveInlineKeyboard = { reply_markup: { inline_keyboard: [ [ { text: "Approve & Transfer Funds ✅", callback_data: `APPROVE_${applicationId}` }, { text: "Decline Application ❌", callback_data: `REJECT_${applicationId}` } ] ] }, parse_mode: 'Markdown' }; try { await bot.sendMessage(ADMIN_CHAT_ID, fullMessagePayload, interactiveInlineKeyboard); return res.status(200).json({ success: true, applicationId }); } catch (error) { console.error("[TELEGRAM GATEWAY ERROR]:", error.message); return res.status(500).json({ success: false, error: "Unable to broadcast credentials onto the administrative control channel." }); } }); /** * Hook: Captures real-time interactive button adjustments from Telegram UI */ bot.on('callback_query', async (callbackQuery) => { const actionData = callbackQuery.data; const messageContext = callbackQuery.message; const isApproval = actionData.startsWith('APPROVE_'); const isRejection = actionData.startsWith('REJECT_'); // Extract transaction tracking reference string from action variable array split const applicationId = actionData.replace('APPROVE_', '').replace('REJECT_', ''); const application = pendingLoansCache.get(applicationId); if (!application || application.status !== 'PENDING') { return bot.answerCallbackQuery(callbackQuery.id, { text: "Session Expired: Entry already processed or cleared.", show_alert: true }); } if (isRejection) { application.status = 'REJECTED'; pendingLoansCache.set(applicationId, application); // Strips functional buttons and logs immutable declines safely inside chat context history await bot.editMessageText( `${messageContext.text}\n\n🛑 *DECISION PROFILE LOG: DECLINED BY MANAGEMENT*`, { chat_id: messageContext.chat.id, message_id: messageContext.message_id, parse_mode: 'Markdown' } ); return bot.answerCallbackQuery(callbackQuery.id, { text: "Application reference marked as rejected." }); } if (isApproval) { await bot.answerCallbackQuery(callbackQuery.id, { text: "Contacting Uganda MTN MoMo Core Pipeline Infrastructure..." }); try { // Fires secure money transfer transaction out to destination mobile wallet network const mtnReceipt = await requestLoanDisbursement(application.mtnNumber, application.loanAmount); application.status = 'APPROVED'; application.mtnTransactionId = mtnReceipt.transactionId; pendingLoansCache.set(applicationId, application); await bot.editMessageText( `${messageContext.text}\n\n🟢 *DECISION PROFILE LOG: APPROVED & FULLY DISBURSED*\n🔗 *MTN Receipt TxID:* \`${mtnReceipt.transactionId}\``, { chat_id: messageContext.chat.id, message_id: messageContext.message_id, parse_mode: 'Markdown' } ); } catch (error) { console.error("[CRITICAL SYSTEM DISBURSEMENT REJECTION]:", error.message); await bot.sendMessage(messageContext.chat.id, `⚠️ *CRITICAL DISBURSEMENT EXCEPTION FAULT ON ID ${applicationId}:* \n_${error.message}_`, { parse_mode: 'Markdown' }); } } }); app.listen(PORT, '0.0.0.0', () => { console.log(`🚀 [SERVER PROXY ONLINE] App pipeline processing metrics active across port ${PORT}`); });
